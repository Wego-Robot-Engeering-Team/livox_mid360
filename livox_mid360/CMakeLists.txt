cmake_minimum_required(VERSION 3.16)
project(livox_mid360)
# Use the Python ABI of Ubuntu's binary ROS distribution by default.
if(NOT DEFINED Python3_EXECUTABLE)
  set(Python3_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "Python interpreter for ROS")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(leveling_core src/leveling_core.cpp)
target_include_directories(leveling_core PUBLIC include)
target_link_libraries(leveling_core Eigen3::Eigen)
ament_target_dependencies(leveling_core sensor_msgs)
target_compile_options(leveling_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(cloud_leveling_node src/cloud_leveling_node.cpp)
target_link_libraries(cloud_leveling_node leveling_core)
ament_target_dependencies(cloud_leveling_node rclcpp sensor_msgs geometry_msgs tf2_ros)
add_executable(imu_visualizer_node src/imu_visualizer_node.cpp)
ament_target_dependencies(imu_visualizer_node rclcpp sensor_msgs geometry_msgs visualization_msgs)
foreach(node cloud_leveling_node imu_visualizer_node)
  target_compile_options(${node} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
install(TARGETS cloud_leveling_node imu_visualizer_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS src/measure_process.py DESTINATION lib/${PROJECT_NAME} RENAME measure_process)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_leveling_core test/test_leveling_core.cpp)
  target_link_libraries(test_leveling_core leveling_core)
  ament_add_pytest_test(test_leveling_integration test/test_leveling_integration.py
    ENV "LIVOX_LEVELING_EXECUTABLE=$<TARGET_FILE:cloud_leveling_node>" "ROS_DOMAIN_ID=86"
    TIMEOUT 30)
endif()
ament_package()
